import base64 from pathlib import Path def get_base64_image(filename, mime_type): """Reads a local image and converts it to a Base64 string for HTML embedding.""" # Check both the current directory and the /mnt/data folder for path in [Path(filename), Path(f'/mnt/data/{filename}')]: if path.exists(): with open(path, "rb") as img_file: encoded = base64.b64encode(img_file.read()).decode('utf-8') return f"data:{mime_type};base64,{encoded}" # Fallback to just the filename if the script can't find it return filename # Generate Base64 strings from your uploaded files bg_b64 = get_base64_image("image_cdfa7b.jpg", "image/jpeg") logo_b64 = get_base64_image("image_cdfafc.png", "image/png") html_template = r''' Live Bracket — Clash Royale Championship
Live Tournament Bracket

Enter Challengers

Summon the 16 players to generate the grand bracket.

ARENA 1: Round of 16
🏆

Grand Champion

Name
''' # Safely inject the Base64 image data directly into the HTML payload final_html = html_template.replace('__BACKGROUND_IMAGE__', bg_b64).replace('__LOGO_IMAGE__', logo_b64) output_path = Path("/mnt/data/clash_royale_championship_live.html") output_path.write_text(final_html, encoding="utf-8") print(f"Updated successfully at: {output_path}. The background and logo are now permanently embedded!")
